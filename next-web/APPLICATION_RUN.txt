main()
  └─ SpringApplication.run()
       ├─ new SpringApplication()  推断类型/加载Initializer/Listener
       └─ run()
            ├─ prepareEnvironment()          加载配置、发布Environment事件
            ├─ createApplicationContext()    创建IoC容器
            ├─ prepareContext()              注册主类、执行Initializer
            ├─ refreshContext()              ★ 刷新容器
            │    ├─ invokeBeanFactoryPostProcessors()  自动配置
            │    ├─ registerBeanPostProcessors()
            │    ├─ onRefresh()                        启动内嵌Web服务器
            │    ├─ finishBeanFactoryInitialization()  实例化Bean
            │    └─ finishRefresh()
            ├─ afterRefresh()
            ├─ callRunners()                 执行ApplicationRunner/CommandLineRunner
            └─ 发布ApplicationReadyEvent



   // 1. 启动计时器、创建 BootstrapContext
    // 2. 获取 SpringApplicationRunListeners（发布启动事件）
    // 3. 准备环境 prepareEnvironment
    // 4. 打印 Banner
    // 5. 创建 ApplicationContext
    // 6. 准备上下文 prepareContext
    // 7. 刷新上下文 refreshContext  ← 核心
    // 8. 后置处理 afterRefresh
    // 9. 发布 started 事件、执行 Runner
    // 10. 发布 ready 事件、返回上下文
